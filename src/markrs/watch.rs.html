12/// Starts the watch mode, which includes:
13/// 1. A file watcher that triggers a rebuild on changes.
14/// 2. A local web server that serves the output directory.
15/// 3. A WebSocket server that sends reload signals to the browser.
16///
17/// # Arguments
18/// * `input_dir` - Directory to watch for changes
19/// * `output_dir` - Directory where output is served from
20/// * `full_build_fn` - Called for initial build
21/// * `incremental_build_fn` - Called with changed file paths for incremental rebuilds
22pub fn start_watch_mode<F, I>(
23    input_dir: String,
24    output_dir: String,
25    full_build_fn: F,
26    incremental_build_fn: I,
27) -> Result<(), Box<dyn std::error::Error + Send + Sync>>
28where
29    F: Fn() -> Result<(), Box<dyn std::error::Error + Send + Sync>> + Send + Sync + 'static,
30    I: Fn(&[PathBuf]) -> Result<(), Box<dyn std::error::Error + Send + Sync>>
31        + Send
32        + Sync
33        + 'static,
34{
35    info!("Starting watch mode...");
36    info!("Serving at http://localhost:3030");
37
38    // Create a Tokio runtime for the async server and watcher components
39    let rt = Runtime::new()?;
40
41    rt.block_on(async {
42        let (tx, _rx) = broadcast::channel::<String>(16);
43
44        // Channel for file watcher events -> debouncer loop (now passes paths)
45        let (notify_tx, mut notify_rx) = tokio::sync::mpsc::unbounded_channel::<PathBuf>();
46
47        // Shared set to collect changed paths during debounce window
48        let pending_paths: Arc<Mutex<HashSet<PathBuf>>> = Arc::new(Mutex::new(HashSet::new()));
49        let pending_paths_watcher = Arc::clone(&pending_paths);
50
51        // 1. Setup File Watcher
52        let full_build_fn = Arc::new(full_build_fn);
53        let incremental_build_fn = Arc::new(incremental_build_fn);
54        let input_path = input_dir.clone();
55
56        // Run initial full build
57        if let Err(e) = full_build_fn() {
58            error!("Initial build failed: {}", e);
59        }
60
61        let mut watcher = RecommendedWatcher::new(
62            move |res: Result<Event, notify::Error>| {
63                match res {
64                    Ok(event) => {
65                        // Filter to only relevant events: Create, Modify (data changes)
66                        // Ignore: Access, Remove, metadata-only changes
67                        let dominated = matches!(
68                            event.kind,
69                            EventKind::Create(_)
70                                | EventKind::Modify(notify::event::ModifyKind::Data(_))
71                                | EventKind::Modify(notify::event::ModifyKind::Any)
72                        );
73
74                        if dominated {
75                            for path in event.paths {
76                                info!("File changed: {:?}", path);
77                                // Add to pending set and notify
78                                if let Ok(mut paths) = pending_paths_watcher.lock() {
79                                    paths.insert(path.clone());
80                                }
81                                let _ = notify_tx.send(path);
82                            }
83                        }
84                    }
85                    Err(e) => error!("Watch error: {:?}", e),
86                }
87            },
88            Config::default(), // Use native watcher (no poll interval for faster events on Windows)
89        )?;
90
91        watcher.watch(Path::new(&input_path), RecursiveMode::Recursive)?;
92
93        // Spawn logic to handle debounced rebuilding
94        let builder_tx = tx.clone();
95        let incremental_build_fn_clone = Arc::clone(&incremental_build_fn);
96        let pending_paths_builder = Arc::clone(&pending_paths);
97        tokio::spawn(async move {
98            loop {
99                // Wait for the first notification
100                if notify_rx.recv().await.is_none() {
101                    break;
102                }
103
104                // Debounce: Wait a short buffer for more events to trickle in (e.g., "save all")
105                tokio::time::sleep(Duration::from_millis(50)).await;
106
107                // Drain any pending events that happened during the wait
108                while notify_rx.try_recv().is_ok() {}
109
110                // Collect and clear pending paths
111                let changed_paths: Vec<PathBuf> = {
112                    let mut paths = pending_paths_builder.lock().unwrap();
113                    let collected: Vec<PathBuf> = paths.drain().collect();
114                    collected
115                };
116
117                if changed_paths.is_empty() {
118                    continue;
119                }
120
121                info!("Rebuilding {} file(s)...", changed_paths.len());
122
123                // Offload blocking build to a dedicated thread to avoid starving the async runtime
124                let build_fn = Arc::clone(&incremental_build_fn_clone);
125                let result = tokio::task::spawn_blocking(move || build_fn(&changed_paths)).await;
126
127                match result {
128                    Ok(Ok(_)) => {
129                        info!("Rebuild successful.");
130                        // Send reload signal
131                        let receiver_count = builder_tx.receiver_count();
132                        info!(
133                            "Broadcasting reload to {} connected clients...",
134                            receiver_count
135                        );
136                        match builder_tx.send("reload".to_string()) {
137                            Ok(n) => info!("Reload signal sent to {} receivers", n),
138                            Err(e) => error!("Failed to send reload signal: {:?}", e),
139                        }
140                    }
141                    Ok(Err(e)) => {
142                        error!("Rebuild failed: {}", e);
143                    }
144                    Err(e) => {
145                        error!("Join error: {}", e);
146                    }
147                }
148            }
149        });
150
151        // 2. Setup Web Server and WebSockets
152
153        // Serve static files with no-cache headers to ensure browser sees updates
154        let static_files = warp::fs::dir(output_dir.clone()).map(|reply| {
155            warp::reply::with_header(reply, "Cache-Control", "no-store, must-revalidate")
156        });
157
158        // WebSocket route
159        let ws_route = warp::path("ws")
160            .and(warp::ws())
161            .and(warp::any().map(move || tx.clone()))
162            .map(|ws: warp::ws::Ws, tx: broadcast::Sender<String>| {
163                ws.on_upgrade(move |socket| handle_ws_client(socket, tx))
164            });
165
166        // Combine routes
167        let routes = ws_route.or(static_files);
168
169        warp::serve(routes).run(([127, 0, 0, 1], 3030)).await;
170
171        Ok(())
172    })
173}

175async fn handle_ws_client(ws: warp::ws::WebSocket, tx: broadcast::Sender<String>) {
176    use futures_util::{SinkExt, StreamExt};
177
178    info!(
179        "WebSocket client connected. Total receivers: {}",
180        tx.receiver_count()
181    );
182    let mut rx = tx.subscribe();
183    info!(
184        "Subscribed to broadcast channel. Now {} receivers.",
185        tx.receiver_count()
186    );
187
188    // Split WebSocket into sender and receiver
189    let (mut ws_tx, mut ws_rx) = ws.split();
190
191    loop {
192        tokio::select! {
193            // Handle broadcast messages (reload signals)
194            broadcast_result = rx.recv() => {
195                match broadcast_result {
196                    Ok(msg) if msg == "reload" => {
197                        info!("Sending reload signal to browser...");
198                        if let Err(e) = ws_tx.send(warp::ws::Message::text("reload")).await {
199                            error!("Failed to send reload to WebSocket: {:?}", e);
200                            break;
201                        }
202                        // Flush immediately for fastest delivery
203                        if let Err(e) = ws_tx.flush().await {
204                            error!("Failed to flush WebSocket: {:?}", e);
205                            break;
206                        }
207                        info!("Reload signal delivered to browser.");
208                    }
209                    Ok(_) => {} // Ignore other messages
210                    Err(e) => {
211                        error!("Broadcast channel error: {:?}", e);
212                        break;
213                    }
214                }
215            }
216            // Handle incoming WebSocket messages (keep connection alive)
217            ws_msg = ws_rx.next() => {
218                match ws_msg {
219                    Some(Ok(msg)) => {
220                        // Just acknowledge we received something - keeps connection alive
221                        if msg.is_close() {
222                            info!("WebSocket client disconnected.");
223                            break;
224                        }
225                        // Ping/pong is handled automatically by warp
226                    }
227                    Some(Err(e)) => {
228                        error!("WebSocket receive error: {:?}", e);
229                        break;
230                    }
231                    None => {
232                        info!("WebSocket stream ended.");
233                        break;
234                    }
235                }
236            }
237        }
238    }
